\section{Course Content}%
\label{sec:CourseContent}%
\subsection{Unit I: Introduction to Algorithms (10 Hours)}%
\label{subsec:UnitIIntroductiontoAlgorithms(10Hours)}%
\begin{itemize}%
\item Algorithm definition and characteristics, Space complexity, Time complexity- worst case, best 1 1,3%
\item case, average case, Complexity, asymptotic notation, Recursive and non-recursive algorithms,%
\item Sorting algorithms : insertion sort, heap sort, bubble sort, Sorting in linear time: counting sort,%
\item Searching algorithms: Linear, Binary%
\end{itemize}

%
\subsection{Unit II: Divide and Conquer strategy and Greedy (14 Hours)}%
\label{subsec:UnitIIDivideandConquerstrategyandGreedy(14Hours)}%
\begin{itemize}%
\item General method, control abstraction, Binary search, Merge sort, Quick sort, Comparison%
\item between Traditional Method of Matrix Multiplication vs. Strassen‟s Matrix Multiplication,%
\item Writing simple algorithm using Divide and conquer strategy: power(x,n), find occurrence of a%
\item number from array of N integers, to find minimum from an array, mini-max algorithm, largest%
\item number multiplication, simple convex algorithm%
\item Greedy Method: Knapsack problem, Job sequencing with deadlines, Minimum-cost spanning%
\item trees: Kruskal and Prim‟s algorithm, Optimal storage on tapes, Optimal merge patterns,%
\item Huffman coding, Shortest Path :Dijkstra‟s Algorithm%
\end{itemize}

%
\subsection{Unit III: Dynamic Programming and Decrease and (16 Hours)}%
\label{subsec:UnitIIIDynamicProgrammingandDecreaseand(16Hours)}%
\begin{itemize}%
\item Principle of optimality, Matrix chain multiplication, 0/1 Knapsack Problem i)Merge \& Purge%
\item ii)Functional Method, Bellman Ford Algorithm, All pairs Shortest Path Floyd-Warshall%
\item Algorithm, Travelling Salesperson problem, Longest common subsequence, String editing%
\item Decrease and conquer: Definition of Graph Representation of Graph, By Constant - DFS and%
\item BFS, Topological sorting, Strongly Connected components and spanning trees, Articulation%
\item Point and Bridge edge%
\end{itemize}

%
\subsection{Unit IV: Backtracking, Branch and Bound (16 Hours)}%
\label{subsec:UnitIVBacktracking,BranchandBound(16Hours)}%
\begin{itemize}%
\item General method, Fixed Tuple vs. Variable Tuple Formulation, n- Queen‟s problem, Graph%
\item Branch and Bound Techniques: Introduction : Branch and bound terms like definition of live%
\item node, E-node, Dead node, Least cost (LC) search, Least cost Branch and Bound (LCBB)%
\item 0/1 knapsack problem using LCBB method (fixed tuple size), Travelling Salesman problem%
\item using LCBB method (variable tuple size)%
\end{itemize}

%
\subsection{Unit V: Problem Classification (4 Hours)}%
\label{subsec:UnitVProblemClassification(4Hours)}%
\begin{itemize}%
\item The class of P, NP, NP-hard and NP –Complete, Relationship among P%
\item class, NP class, NP-hard and NP –Complete, Cook‟s theorem%
\end{itemize}
